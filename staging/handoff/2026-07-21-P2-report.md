## Reproducible Stats Command

```powershell
@'
from pathlib import Path
from collections import Counter
import yaml
ROOT = Path(r"<repo-root>")
levels = Counter()
filled = Counter()
tiers = Counter()
refs = Counter()
topic_total = 0
for path in [ROOT / "curriculum" / "topics" / "a1.yaml", ROOT / "curriculum" / "topics" / "a2.yaml"]:
    data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
    for topic in data["topics"]:
        key = (topic["cefr"], topic["track"])
        levels[key] += 1
        topic_total += 1
        if all(field in topic for field in [
            "dimensions", "mastery_criteria", "typical_errors", "examples",
            "contexts", "lexicon", "explanation_language",
        ]):
            filled[key] += 1
        if topic["track"] == "grammar-engine":
            tiers[topic["frequency_tier"]] += 1
        refs.update(topic.get("lexicon", []))
lexicon = {}
for path in sorted((ROOT / "curriculum" / "lexicon").glob("*.yaml")):
    if path.name.startswith("_"):
        continue
    data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
    for item in data.get("lexical_items", []):
        lexicon[item["id"]] = item
priority_order = {"CORE": 0, "HIGH": 1, "USEFUL": 2, "SPECIALIZED": 3, "INCIDENTAL": 4}
unused = [item for item_id, item in lexicon.items() if item_id not in refs]
unused_sorted = sorted(
    unused,
    key=lambda item: (
        priority_order.get(item.get("curriculum_priority_band"), 99),
        item.get("cefr", ""),
        item["id"],
    ),
)
print(f"topics_total={topic_total}")
print(f"topics_filled={sum(filled.values())}")
print("coverage_by_level_track:")
for key in sorted(levels):
    print(f"  {key[0]} {key[1]}: {filled[key]}/{levels[key]}")
print("frequency_tiers:")
for tier, count in sorted(tiers.items()):
    print(f"  {tier}: {count}")
print(f"lexicon_total={len(lexicon)}")
print(f"lexicon_used={len(refs)}")
print(f"lexicon_refs_total={sum(refs.values())}")
print(f"lexicon_unused={len(lexicon) - len(refs)}")
print("top_unused_priority_ranked:")
for item in unused_sorted[:25]:
    print(f"  {item['id']} ({item.get('cefr')}, {item.get('curriculum_priority_band')}, {item.get('type')})")
'@ | .\.venv\Scripts\python.exe -
```
